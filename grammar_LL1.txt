program -> declaration_list

declaration_list -> declaration sep_semicolon declaration_list
                  | ε

sep_semicolon -> SEMICOLON
               | D_SEMICOLON

declaration -> function_decl
             | type_decl
             | protocol_decl
             | stmt

function_decl -> KW_FUNCTION IDENTIFIER LPAREN opt_param_list RPAREN opt_type_annotation function_body

opt_param_list -> param_list
                | ε

param_list -> param param_list_tail
param_list_tail -> COMMA param param_list_tail
                 | ε

param -> IDENTIFIER opt_type_annotation

function_body -> ARROW expression
               | block_expr

type_decl -> KW_TYPE IDENTIFIER type_params type_inheritance LBRACE type_body RBRACE

type_params -> LPAREN t_param_list RPAREN
             | ε

t_param_list -> t_param t_param_list_tail
t_param_list_tail -> COMMA t_param t_param_list_tail
                   | ε

t_param -> IDENTIFIER opt_type_annotation

type_inheritance -> KW_INHERITS IDENTIFIER parent_args
                  | ε

parent_args -> LPAREN args RPAREN
             | ε

type_body -> member type_body
           | ε

member -> IDENTIFIER member_tail
member_tail -> opt_type_annotation opt_initializer SEMICOLON
             | LPAREN opt_param_list RPAREN opt_type_annotation function_body sep_semicolon

opt_initializer -> initializer
                 | ε

protocol_decl -> KW_PROTOCOL IDENTIFIER protocol_inheritance LBRACE protocol_body RBRACE

protocol_inheritance -> KW_EXTENDS IDENTIFIER
                      | ε

protocol_body -> protocol_method protocol_body_tail
protocol_body_tail -> SEMICOLON protocol_method protocol_body_tail
                    | ε

protocol_method -> IDENTIFIER LPAREN method_params RPAREN type_annotation SEMICOLON

method_params -> m_param_list
               | ε

m_param_list -> m_param m_param_list_tail
m_param_list_tail -> COMMA m_param m_param_list_tail
                   | ε

m_param -> IDENTIFIER type_annotation

expression -> let_expr
            | if_expr
            | while_expr
            | block_expr
            | for_expr
            | assignment

block_expr -> LBRACE expression_list RBRACE

expression_list -> opt_expression expression_list_tail
expression_list_tail -> SEMICOLON opt_expression expression_list_tail
                      | ε

opt_expression -> expression
                | ε

let_expr -> KW_LET binding_list KW_IN expression

binding -> IDENTIFIER opt_type_annotation initializer

binding_list -> binding binding_list_tail
binding_list_tail -> COMMA binding binding_list_tail
                   | ε

if_expr -> KW_IF LPAREN expression RPAREN expression elif_list KW_ELSE expression
elif_list -> KW_ELIF LPAREN expression RPAREN expression elif_list
           | ε

while_expr -> KW_WHILE LPAREN expression RPAREN expression
for_expr -> KW_FOR LPAREN IDENTIFIER opt_type_annotation KW_IN expression RPAREN expression

let_expr -> KW_LET binding_list KW_IN expression

binding -> IDENTIFIER opt_type_annotation initializer

binding_list -> binding binding_list_tail
binding_list_tail -> COMMA binding binding_list_tail
                   | ε

if_expr -> KW_IF LPAREN expression RPAREN expression elif_list KW_ELSE expression
elif_list -> KW_ELIF LPAREN expression RPAREN expression elif_list
           | ε

while_expr -> KW_WHILE LPAREN expression RPAREN expression
for_expr -> KW_FOR LPAREN IDENTIFIER opt_type_annotation KW_IN expression RPAREN expression

assignment -> or_expr assignment_tail
assignment_tail -> OP_DESTRUCT_ASSIGN or_expr assignment_tail
                 | ε

or_expr -> and_expr or_expr_tail
or_expr_tail -> OP_OR and_expr or_expr_tail
              | ε

and_expr -> equality and_expr_tail
and_expr_tail -> OP_AND equality and_expr_tail
               | ε

equality -> comparison equality_tail
equality_tail -> OP_NOT_EQUAL comparison equality_tail
               | OP_EQUAL comparison equality_tail
               | ε

comparison -> term comparison_tail
comparison_tail -> OP_GREATER term comparison_tail
                 | OP_GREATER_EQ term comparison_tail
                 | OP_LESS term comparison_tail
                 | OP_LESS_EQ term comparison_tail
                 | ε

term -> factor term_tail
term_tail -> OP_MINUS factor term_tail
           | OP_PLUS factor term_tail
           | ε

factor -> unary factor_tail
factor_tail -> OP_DIVIDE unary factor_tail
             | OP_MODULE unary factor_tail
             | OP_MULTIPLY unary factor_tail
             | OP_EXPONENT unary factor_tail
             | OP_CONCAT unary factor_tail
             | OP_DOBLE_CONCAT unary factor_tail
             | ε

unary -> OP_NOT unary
       | OP_MINUS unary
       | var_call

primary -> FALSE
         | TRUE
         | T_NULL
         | NUMBER
         | STRING
         | IDENTIFIER finish_call
         | LPAREN expression RPAREN
         | KW_NEW IDENTIFIER LPAREN args RPAREN

var_call -> primary var_call_tail
var_call_tail -> DOT IDENTIFIER finish_call var_call_tail
               | ε

finish_call -> LPAREN args RPAREN
             | ε

args -> expr_list
      | ε

expr_list -> expression expr_list_tail
expr_list_tail -> COMMA expression expr_list_tail
                | ε
